cmake_minimum_required(VERSION 3.8)
project(robot_perception)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(POLICY CMP0074)
  cmake_policy(SET CMP0074 NEW)
endif()
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
set(dependencies rclcpp robot_task_interfaces moveit_msgs sensor_msgs shape_msgs
  geometry_msgs pcl_conversions pcl_ros tf2_ros tf2_eigen)
foreach(dep ${dependencies})
  find_package(${dep} REQUIRED)
endforeach()
find_package(PCL REQUIRED COMPONENTS common features filters io kdtree search segmentation)
add_library(robot_perception_processing SHARED
  src/cluster_extraction.cpp src/object_segmentation.cpp src/plane_segmentation.cpp
  src/normals_curvature_and_rsd_estimation.cpp)
target_include_directories(robot_perception_processing PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include> ${PCL_INCLUDE_DIRS})
target_compile_definitions(robot_perception_processing PUBLIC PCL_NO_PRECOMPILE)
ament_target_dependencies(robot_perception_processing ${dependencies})
target_link_libraries(robot_perception_processing ${PCL_LIBRARIES})
add_executable(get_planning_scene_server src/get_planning_scene_server.cpp)
ament_target_dependencies(get_planning_scene_server ${dependencies})
target_link_libraries(get_planning_scene_server robot_perception_processing)
install(TARGETS robot_perception_processing get_planning_scene_server
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/ai_planning_scene_server DESTINATION lib/${PROJECT_NAME})
install(FILES requirements-ai.txt DESTINATION share/${PROJECT_NAME})
ament_package()
